2. 最终 
```powershell
# 中文环境初始化
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8
$Host.UI.RawUI.WindowTitle = "HF镜像替换工具"

# 管理员权限检测与提权
function Test-Admin {
    $currentUser = [Security.Principal.WindowsIdentity]::GetCurrent()
    (New-Object Security.Principal.WindowsPrincipal $currentUser).IsInRole(
        [Security.Principal.WindowsBuiltInRole]::Administrator
    )
}

if (-not (Test-Admin)) {
    Write-Host "正在请求管理员权限..." -ForegroundColor Yellow
    Start-Process powershell.exe -ArgumentList (
        "-NoProfile -ExecutionPolicy Bypass -File `"$($MyInvocation.MyCommand.Path)`""
    ) -Verb RunAs
    exit
}

Write-Host "√ 已获得管理员权限`n" -ForegroundColor Green

# 配置参数
$ConfigFile = Join-Path $PSScriptRoot "config.ini"
$DefaultInstallDir = Join-Path $env:LOCALAPPDATA "LM-Studio"
$ProcessName = "LM Studio"

# 修复的配置文件处理逻辑
if (Test-Path $ConfigFile) {
    # 过滤掉包含方括号的行并转换为哈希表
    $config = Get-Content $ConfigFile | 
              Where-Object { $_ -notmatch '^\s*\[' } |
              Out-String | 
              ConvertFrom-StringData
    $InstallDir = $config.InstallPath
}

if (-not $InstallDir) {
    $input = Read-Host "请输入安装路径（默认：$DefaultInstallDir）"
    $InstallDir = if ($input) { $input } else { $DefaultInstallDir }
    
    # 生成新的配置文件格式（不带[Settings]节）
    @"
InstallPath=$InstallDir
"@ | Out-File $ConfigFile -Encoding UTF8
    Write-Host "配置已保存到 $ConfigFile" -ForegroundColor Green
}

# 路径验证
if (-not (Test-Path $InstallDir)) {
    Write-Host "错误：路径 '$InstallDir' 无效" -ForegroundColor Red
    exit 1
}

# 进程管理
if ($proc = Get-Process $ProcessName -ErrorAction SilentlyContinue) {
    Write-Host "正在终止进程: $($proc.Name) (PID: $($proc.Id))..."
    $proc | Stop-Process -Force
    Start-Sleep -Seconds 2
}

# 操作选择
do {
    $action = Read-Host @"
请选择操作：
1. 替换 huggingface.co -> hf-mirror.com
2. 还原 hf-mirror.com -> huggingface.co
输入选项 (1/2)
"@
} while ($action -notin 1,2)

$search, $replace = @(
    ("huggingface.co", "hf-mirror.com"),
    ("hf-mirror.com", "huggingface.co")
)[$action-1]

# 文件替换核心逻辑
$count = 0
Get-ChildItem $InstallDir -Recurse -File | ForEach-Object {
    try {
        $content = [IO.File]::ReadAllText($_.FullName, [Text.Encoding]::UTF8)
        if ($content -match $search) {
            $newContent = $content -replace $search, $replace
            [IO.File]::WriteAllText($_.FullName, $newContent, [Text.Encoding]::UTF8)
            $count++
            Write-Host "已更新: $($_.FullName)" -ForegroundColor Cyan
        }
    }
    catch {
        Write-Host "跳过只读文件: $($_.FullName)" -ForegroundColor DarkGray
    }
}

# 操作日志（追加到配置文件）
@"

操作时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
处理文件数: $count
目标目录: $InstallDir
"@ | Out-File $ConfigFile -Append -Encoding UTF8

Write-Host "`n操作完成！共更新 $count 个文件" -ForegroundColor Green
Pause
```
